vob_consume_ui_request = {
    scope = country
    ai_is_valid = { always = no }
    is_shown = { is_player = yes }
    saved_scopes = { language }
    is_valid = { is_player = yes has_variable = vob_open_requested OR = { scope:language = 1 scope:language = 2 } }
    effect = { if = { limit = { is_player = yes has_variable = vob_open_requested OR = { scope:language = 1 scope:language = 2 } }

# Real game-language changes select that language's default index. First use
# in Chinese preserves a migrated 0.1.7 manual English-index preference.
if = { limit = { has_variable = vob_ui_language has_variable = vob_initial_language }
    if = { limit = { NOT = { var:vob_ui_language = 0 } NOT = { var:vob_ui_language = scope:language } }
        if = { limit = { NOT = { var:vob_initial_language = scope:language } }
            set_variable = { name = vob_initial_language value = scope:language }
            set_variable = { name = vob_filter_letter value = 0 }
        }
    }
    if = { limit = { scope:language = 2 NOT = { var:vob_initial_language = 2 } }
        set_variable = { name = vob_initial_language value = 2 }
        set_variable = { name = vob_filter_letter value = 0 }
    }
    set_variable = { name = vob_ui_language value = scope:language }
    vob_refresh_all = yes
}

set_variable = { name = vob_ui_ready value = 1 }
    } }
}
