# VOB batch mode: initialize money transfers at the native maximum.
# The flag is set only while the VOB batch state machine is active. Outside
# batch mode the original minimum (100, capped by fixed income) is unchanged.
money_transfer = {
    quantity_min_value = {
        value = 100
        max = fixed_income
        min = 0

        if = {
            limit = { has_variable = vob_force_money_max }
            value = fixed_income
            divide = 4
            if = {
                limit = { income_transfer_relative_expenses >= 0.5 }
                max = 100
            }
        }
    }
}
